# This directory can also be configured directly with cmake -S smcTuring.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(smcTuring LANGUAGES CXX)
  include(CTest)
endif()

option(SMCTURING_INLINE_ASM "Include the historical MSVC x86 inline assembly variants" ON)
set(smc_inline FALSE)

if(MSVC AND CMAKE_SYSTEM_NAME STREQUAL "Windows")
  if(NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID MATCHES "^(X86|x64)$")
    message(FATAL_ERROR "The MASM backend supports Win32 and x64 targets.")
  endif()
  enable_language(ASM_MASM)
  add_executable(smcTuring main.cpp tm.asm)
  # Keep assembler options on its source: Visual Studio's mixed-language
  # targets do not preserve target-level COMPILE_LANGUAGE filters for MASM.
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_property(SOURCE tm.asm APPEND PROPERTY COMPILE_OPTIONS /DX64)
  else()
    set_property(SOURCE tm.asm APPEND PROPERTY COMPILE_OPTIONS /safeseh)
    if(SMCTURING_INLINE_ASM)
      set(smc_inline TRUE)
      # The historical inline MOV loop uses EBX and a compiler-owned scratch
      # array. Keep its required frame pointer and disabled optimizer.
      set_property(SOURCE main.cpp APPEND PROPERTY COMPILE_OPTIONS /Od /Oy-)
    endif()
  endif()
  set_property(SOURCE main.cpp APPEND PROPERTY COMPILE_OPTIONS /W3)
elseif(CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$" AND CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
  if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang$")
      message(FATAL_ERROR "The macOS assembly backend requires Apple Clang or Clang.")
    endif()
    if(CMAKE_OSX_ARCHITECTURES)
      if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
        message(FATAL_ERROR "The macOS assembly backend targets x86_64; select -DCMAKE_OSX_ARCHITECTURES=x86_64.")
      endif()
    elseif(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
      message(FATAL_ERROR "Select -DCMAKE_OSX_ARCHITECTURES=x86_64 for the Intel macOS backend.")
    endif()
  elseif(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|i[3-6]86|x86)$")
    message(FATAL_ERROR "The Linux assembly backend supports x86 and x86-64 targets.")
  endif()
  enable_language(ASM)
  add_executable(smcTuring main.cpp tm.S)
  target_compile_options(smcTuring PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
else()
  message(FATAL_ERROR "Supported toolchains: Windows MSVC/MASM, Linux GCC/Clang or Intel macOS Clang.")
endif()

if(NOT smc_inline)
  set_property(SOURCE main.cpp APPEND PROPERTY COMPILE_DEFINITIONS SMCTURING_NO_INLINE_ASM)
endif()
target_compile_features(smcTuring PRIVATE cxx_std_11)
set_target_properties(smcTuring PROPERTIES CXX_EXTENSIONS OFF)

if(BUILD_TESTING)
  function(smc_addition_test name input expected iterations)
    add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
      "-DPROGRAM=$<TARGET_FILE:smcTuring>" "-DINPUT=${input}" "-DEXPECTED=${expected}"
      "-DINLINE_VARIANTS=${smc_inline}" "-DITERATIONS=${iterations}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckDemo.cmake")
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
  endfunction()
  smc_addition_test(addition.original 0100_1001_ 1101_0000_ 0)
  smc_addition_test(addition.zero 0000_0000_ 0000_0000_ 0)
  smc_addition_test(addition.increment 0010_0001_ 0011_0000_ 0)
  smc_addition_test(addition.carry 0111_0001_ 1000_0000_ 0)
  smc_addition_test(addition.maximum 0000_1111_ 1111_0000_ 0)
  smc_addition_test(addition.single_bit 0_1_ 1_0_ 0)
  smc_addition_test(benchmark.smoke 0100_1001_ 1101_0000_ 2)
  add_test(NAME input.malformed COMMAND smcTuring --input 0100_bad_ --no-benchmark)
  add_test(NAME input.overflow COMMAND smcTuring --input 1111_0001_ --no-benchmark)
  add_test(NAME input.negative_iterations COMMAND smcTuring --iterations -1)
  add_test(NAME input.missing_option_value COMMAND smcTuring --input)
  set_tests_properties(input.malformed input.overflow input.negative_iterations input.missing_option_value
    PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
  add_test(NAME cli.help COMMAND smcTuring --help)
  set_tests_properties(cli.help PROPERTIES TIMEOUT 10)
endif()

install(TARGETS smcTuring RUNTIME DESTINATION bin)
